\begin{frame}{From ``How Much'' to ``Who Decides''}
  Earlier lectures told us \emph{how much} of a public good is efficient (the Samuelson rule) and \emph{how} to evaluate a specific project (cost-benefit analysis). In the real world, economists don't get to simply implement those answers.
  \begin{itemize}
    \item Public decisions are made through a \textbf{political system}: a single ruler, elected officials, or the direct votes of citizens.
    \item This chapter asks the fourth question of public finance head-on: \textbf{why} do governments do what they do?
  \end{itemize}
  \vspace{0.4em}
  \small We start with the best case --- a government that perfectly measures and aggregates preferences --- then work toward more realistic, messier cases.
\end{frame}

\section{9.1 Unanimous Consent: Lindahl Pricing}

\begin{frame}{Lindahl Pricing: The Idea}
  \begin{block}{Lindahl pricing}
    Individuals honestly report their willingness to pay for a public good, and the government charges each of them exactly that amount to finance it.
  \end{block}
  \begin{itemize}
    \item Proposed by Swedish economist Erik Lindahl in 1919.
    \item Uses each person's \textbf{marginal willingness to pay} --- exactly like an ordinary demand curve, but for a public good.
    \item Corresponds to \textbf{benefit taxation}: you're taxed in proportion to how much you personally value the good.
  \end{itemize}
\end{frame}

\begin{frame}{Worked Example: Ava, Jack, and Fireworks}
  A public good (fireworks, constant marginal cost \$1/unit) is provided to two people, Ava and Jack, in \emph{equal} quantities to both.
  \begin{enumerate}
    \setlength{\itemsep}{2pt}
    \item Government announces tax \emph{prices} --- e.g., Ava pays 50\textcent, Jack pays 50\textcent\ per firework.
    \item Each person announces how many fireworks they want at those prices.
    \item Repeating this traces out each person's willingness-to-pay schedule.
    \item Government sums these to get an aggregate demand curve for the public good.
    \item Aggregate demand is set equal to marginal cost to find the optimal quantity.
    \item Government finances it by charging each person their own willingness to pay.
  \end{enumerate}
\end{frame}

\begin{frame}{Diagram: The Lindahl Equilibrium}
  \begin{center}
  \begin{tikzpicture}[scale=0.62]
    \draw[-{Latex}] (0,0) -- (7.5,0) node[below, font=\small]{Fireworks};
    \draw[-{Latex}] (0,0) -- (0,6.3) node[left, font=\small]{Willingness to pay (\$)};
    \draw[wulight, thick] (0,1.5) -- (6.5,0);
    \node[below, font=\scriptsize, wulight] at (5.3,0.32) {$D_A$ (Ava)};
    \draw[wugreen, thick] (0,4.5) -- (6.5,0);
    \node[above, font=\scriptsize, wugreen] at (3.3,2.22) {$D_J$ (Jack)};
    \draw[wublue, thick] (0,6) -- (6.5,0);
    \node[above, font=\scriptsize, wublue] at (1.3,4.8) {$D_{A+J}$ (sum)};
    \draw[wured, thick] (0,1.5) -- (6.5,1.5);
    \node[above, font=\scriptsize, wured] at (2.2,1.55) {$MC=\$1$};
    \fill (4.87,1.5) circle (1.8pt);
    \draw[dotted] (4.87,0) -- (4.87,1.5);
    \node[below, font=\scriptsize] at (4.87,-0.15) {$75$};
  \end{tikzpicture}
  \end{center}
  \footnotesize Summing Ava's and Jack's individual demand curves gives $D_{A+J}$, which crosses $MC=\$1$ at 75 fireworks. At the Lindahl equilibrium, Ava pays 25\textcent\ and Jack pays 75\textcent\ per firework --- both are happy, and the government exactly covers its cost. This is also the \textbf{efficient} quantity: it's the same vertical-summation logic as the Samuelson rule.
\end{frame}

\begin{frame}{Why Lindahl Pricing Fails in Practice}
  \small
  \begin{enumerate}
    \setlength{\itemsep}{4pt}
    \item \textbf{Preference revelation problem:} people have an incentive to \emph{understate} their valuation to lower their tax price --- exactly the free-rider logic from the public-goods lecture. If Jack lies and claims Ava's (lower) preferences, the Lindahl outcome falls to 50 fireworks at 50\textcent\ each: Jack now pays \$25 for two-thirds as many fireworks as before (he was paying \$56.25 for 75) --- he free-rides, and Ava ends up paying \emph{more} (\$25) for \emph{fewer} fireworks (50 instead of 75).
    \item \textbf{Preference knowledge problem:} even honest people often don't know their own valuation --- how much do \emph{you} value national defense?
    \item \textbf{Preference aggregation problem:} canvassing 322 million US citizens individually for their valuation of missiles and tanks is simply infeasible.
  \end{enumerate}
  \vspace{0.3em}
  \begin{alertblock}{Bottom line}
    Lindahl pricing is efficient in theory but unworkable in practice --- which is why real societies use \textbf{voting} instead.
  \end{alertblock}
\end{frame}

\section{9.2 Aggregating Preferences: Voting}

\begin{frame}{Application: Direct Democracy in the US}
  \begin{block}{Direct democracy}
    Individuals vote directly on the policies that affect their lives.
  \end{block}
  \begin{itemize}
    \item \textbf{Town meetings:} a tradition since 1657; Vermont alone held 1,435 town meetings (1970--1998) with over 63,000 attendees --- more than a fifth of residents, on average.
    \item \textbf{Referendum:} citizens vote on a law or amendment the legislature \emph{already} passed. All 50 states allow \emph{legislative} referenda; 24 allow \emph{popular} referenda (triggered by petition).
    \item \textbf{Voter initiative:} citizens place their \emph{own} legislation on the ballot via petition (24 states allow this). Over 8,000 initiatives filed since 1904; 41\% of the $\sim$2,600 that reached a ballot have passed.
  \end{itemize}
  \small California's 1978 Proposition 13 (severely limiting property tax rates) sparked a nationwide ``tax revolt'' and a wave of renewed initiative activity.
\end{frame}

\begin{frame}{Majority Voting: What ``Consistent'' Requires}
  \begin{block}{Majority voting}
    Individual policy options are put to a vote; the option with the most votes is chosen.
  \end{block}
  For majority voting to \emph{consistently} aggregate preferences, it must satisfy three conditions:
  \begin{itemize}
    \item \textbf{Dominance:} if every voter prefers $A$ to $B$, society's choice must be $A$.
    \item \textbf{Transitivity:} if $A \succ B$ and $B \succ C$, then $A \succ C$.
    \item \textbf{Independence of irrelevant alternatives:} adding a third option $C$ shouldn't flip society's ranking of $A$ vs.\ $B$.
  \end{itemize}
\end{frame}

\begin{frame}{Worked Example: When Majority Voting Works}
  A town chooses school funding: High ($H$), Medium ($M$), or Low ($L$). Three equal-sized voter groups:
  \begin{center}
  \small
  \begin{tabular}{@{}lccc@{}}
    \toprule
    & \textbf{Parents} & \textbf{Elders} & \textbf{Young couples} \\
    \midrule
    1st choice & $H$ & $L$ & $M$ \\
    2nd choice & $M$ & $M$ & $L$ \\
    3rd choice & $L$ & $H$ & $H$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.4em}
  Pairwise votes: $L$ beats $H$ (2--1) \quad $M$ beats $H$ (2--1) \quad $M$ beats $L$ (2--1).
  \begin{alertblock}{$M$ wins every pairwise vote}
    Whatever order the votes happen in, medium spending wins --- the preference of the \textbf{median} voter group (young couples).
  \end{alertblock}
\end{frame}

\begin{frame}{Worked Example: When Majority Voting Fails}
  Now replace ``elders'' with parents considering \textbf{private school}: their worst outcome is $M$ (high enough taxes to hurt, too low quality to avoid private school anyway).
  \begin{center}
  \small
  \begin{tabular}{@{}lccc@{}}
    \toprule
    & \textbf{Public-school parents} & \textbf{Private-school parents} & \textbf{Young couples} \\
    \midrule
    1st choice & $H$ & $L$ & $M$ \\
    2nd choice & $M$ & $H$ & $L$ \\
    3rd choice & $L$ & $M$ & $H$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.3em}
  Pairwise votes: $L$ beats $H$ \quad $H$ beats $M$ \quad $M$ beats $L$.
  \begin{alertblock}{A voting cycle}
    $L \succ H \succ M \succ L$: there is \textbf{no} consistent winner. Whoever controls the voting agenda --- the \emph{order} of pairwise votes --- can engineer whichever outcome they want.
  \end{alertblock}
\end{frame}

\begin{frame}{Arrow's Impossibility Theorem}
  The cycling problem above isn't a quirk of majority voting specifically --- Kenneth Arrow (Nobel, 1972) proved something much stronger.
  \begin{itemize}
    \item Simple plurality voting? Just produces a three-way tie (each group's favorite gets equal support).
    \item Weighted voting (3/2/1 points for 1st/2nd/3rd)? Breaks the first example's tie fine --- but in the \emph{second} example, it produces another three-way tie.
  \end{itemize}
  \begin{block}{Arrow's Theorem}
    There is \textbf{no} social decision rule that converts individual preferences into a consistent aggregate decision, except by either (1) restricting the type of preferences allowed, or (2) imposing a dictatorship.
  \end{block}
\end{frame}

\begin{frame}{Single-Peaked Preferences}
  \begin{block}{Single-peaked preferences}
    Utility has only one local maximum (``peak'') --- it falls as choices move away from that peak in \emph{either} direction.
  \end{block}
  \begin{itemize}
    \item Restricting preferences to be single-peaked is the standard way economists escape Arrow's impossibility result.
    \item Usually reasonable for public goods --- but breaks down when a \textbf{private substitute} exists. Private-school parents wanted low spending (afford private school) \emph{or} high spending (excellent public school) --- but \emph{not} the mediocre, still-costly middle ground.
    \item Same logic for a local park: a lovely park, or no park (rely on your backyard) --- but not a mediocre one you still pay taxes for.
  \end{itemize}
\end{frame}

\begin{frame}{The Median Voter Theorem}
  \begin{block}{Median Voter Theorem}
    If preferences are single-peaked, majority voting yields the outcome preferred by the \textbf{median voter} --- the voter whose tastes sit exactly in the middle, with equal numbers preferring more and preferring less.
  \end{block}
  \begin{itemize}
    \item Remarkably convenient: the government only needs to find the \emph{one} median voter and implement their preference --- it doesn't need to know anything about anyone else's preferences.
    \item But convenient $\neq$ efficient, as the next example shows.
  \end{itemize}
\end{frame}

\begin{frame}{The Potential Inefficiency of the Median Voter Outcome}
  Your hometown (1,001 voters) considers a \$40,040 monument to you, financed by a \$40 tax on \emph{everyone}. Preferences are single-peaked, so the median voter decides.
  \begin{itemize}
    \item 500 voters value the monument at \$100 each; 501 voters value it at \$0.
    \item \textbf{Social marginal benefit:} $500(100) + 501(0) = \$50{,}000 > \$40{,}040$ \textbf{social marginal cost} --- building it is efficient.
  \end{itemize}
  \begin{alertblock}{But it loses, 501--500}
    The median voter --- one of the 501 who values it at \$0 --- votes no. The median voter outcome ignores \textbf{intensity} of preferences: it doesn't matter that 500 people wanted it very badly, only that the median voter didn't want it at all.
  \end{alertblock}
\end{frame}

\section{9.3 Representative Democracy}

\begin{frame}{From Direct to Representative Democracy}
  Most developed democracies don't vote directly on public goods --- voters elect \textbf{representatives} who vote on their behalf. The standard theory: politicians are \textbf{vote-maximizers}, so (with single-peaked preferences) they too converge on the median voter's preferred outcome.
\end{frame}

\begin{frame}{Diagram: Vote Maximization Leads to the Median Voter}
  \begin{center}
  \begin{tikzpicture}[scale=0.85]
    \draw[-{Latex}] (0,0) -- (10,0);
    \node[below, font=\tiny] at (0,-0.05) {0\%};
    \node[below, font=\tiny] at (5,-0.05) {25\%};
    \node[below, font=\tiny] at (10,-0.05) {50\%};
    \node[below, font=\small] at (5,-2.05) {Defense spending (\% of budget)};
    \draw[dashed] (5,0) -- (5,1.8);
    \node[above, font=\tiny] at (5,1.9) {median voter};
    \fill[wulight] (1.5,0) circle (2.2pt); \node[above, font=\scriptsize, wulight] at (1.5,0.15) {$J_1$ (Joe, initial)};
    \fill[wured] (8.5,0) circle (2.2pt); \node[above, font=\scriptsize, wured] at (8.5,0.15) {$D_1$ (Donald, initial)};
    \draw[-{Latex}, wulight] (1.5,-0.6) -- (4.8,-0.6);
    \draw[-{Latex}, wured] (8.5,-1.1) -- (5.2,-1.1);
    \fill[wublue] (5,-1.6) circle (2.2pt); \node[below, font=\scriptsize, wublue] at (5,-1.75) {both converge here};
  \end{tikzpicture}
  \end{center}
  \vspace{0.6em}
  \footnotesize (Downs, 1957.) Joe starts appealing to low-defense voters, Donald to high-defense voters --- an even split. Either candidate can win more votes by shifting toward the median. This jockeying continues until \textbf{both} candidates support exactly the median voter's preferred 25\% --- at which point neither can gain by moving.
\end{frame}

\begin{frame}{Assumptions of the Median Voter Model}
  The model is powerful --- but rests on six assumptions, each of which can fail in reality:
  \begin{enumerate}
    \setlength{\itemsep}{2pt}
    \item Voting is single-dimensional.
    \item There are only two candidates.
    \item Politicians care only about votes --- no ideology.
    \item All affected citizens vote.
    \item Money plays no role.
    \item Everyone --- voters and politicians alike --- has full information.
  \end{enumerate}
\end{frame}

\begin{frame}{When the Assumptions Break: Dimensions, Candidates, Ideology}
  \small
  \begin{itemize}
    \item \textbf{Multi-dimensional voting:} real elections bundle many issues. If preferences correlate across issues, voting can still behave as if single-dimensional --- but if not, there may be no stable equilibrium at all.
    \item \textbf{More than two candidates:} with three or more, there is no stable equilibrium --- any candidate can always gain votes by repositioning. (The US, with its two major parties, is a partial exception.)
    \item \textbf{Ideology:} politicians with genuine convictions can shift \emph{voter} views toward their own position, rather than simply chasing the median.
  \end{itemize}
\end{frame}

\begin{frame}{When the Assumptions Break: Turnout, Money, Information}
  \small
  \begin{itemize}
    \item \textbf{Selective voting:} only about half of citizens vote in presidential years, roughly a third in off-years. If the voters who show up are more ideological than the population, politicians rationally target \emph{them}, not the true median.
    \item \textbf{Money:} the cost of winning a US House seat more than doubled 1990--2018 (\$810K $\to$ \$2.09M); a Senate seat, \$6.3M $\to$ \$14.9M. Taking an extreme position that maximizes \emph{fundraising} can serve long-run vote maximization even if it isn't the median position.
    \item \textbf{Imperfect information:} Senator Robert Byrd, asked if he knew what was in a 4,000-page, \$520 billion spending bill: \emph{``Are you kidding? Only God knows what's in this conference report!''}
  \end{itemize}
\end{frame}

\begin{frame}{Lobbying}
  \begin{block}{Lobbying}
    The expending of resources by individuals or groups to influence a politician.
  \end{block}
  Politicians listen to lobbyists for two reasons: they provide \textbf{information} on complex issues, and they \textbf{reward} supportive politicians with votes and campaign funds.
  \vspace{0.4em}
  \small \textbf{The danger:} a project benefiting 100 people by \$1M each but costing 259.9999 million people \$100 each has negative net social benefit ($100\text{M} < \$26\text{B}$) --- yet the concentrated beneficiaries have a strong incentive to lobby, while each diffuse loser has almost no incentive to fight \$100 of harm. A self-interested politician may back it anyway.\\
  \srcnote{Bertrand et al.\ (2014) find lobbyists' value lies as much in \emph{political connections} as in \emph{expertise}.}
\end{frame}

\begin{frame}{Application: Farm Policy in the United States}
  Agriculture employed 35\% of US workers in 1900; today, just 1.3\%. Yet the sector still receives \textbf{\$22.4 billion} in direct federal support each year.
  \begin{itemize}
    \item Only 20 of $\sim$400 US crops are subsidy-eligible, and subsidies scale with output --- so \textbf{75\% of subsidies go to the top 10\% of recipients} (avg.\ \$560K/year), while the bottom 80\% get \$8,014/year.
    \item Cost to the typical American household: $\approx$\$144/year. Gain to the average recipient farm: \$15,736/year --- concentrated benefits, diffuse costs, textbook lobbying success.
  \end{itemize}
  \small Not uniquely American: the EU spends $\approx$\$113B/year on farmers (\$2.33/day per cow); Japan, over \$47B, with $\approx$800\% rice tariffs. New Zealand eliminated most farm subsidies in the mid-1980s; after a bumpy six-year transition, farm profitability and rural employment stabilized at roughly their prior levels.
\end{frame}

\begin{frame}{Evidence on the Median Voter Model}
  \footnotesize
  \textbf{Redistricting studies} directly test whether the median voter matters:
  \begin{itemize}
    \setlength{\itemsep}{1pt}
    \item \textbf{Stratmann (2000):} when decennial redistricting made a district more Republican-leaning (based on presidential voting), the \emph{same} incumbent started voting more conservatively --- strong support for the theory.
    \item \textbf{Aidt \& Shvets (2012):} term-limited legislators (who no longer need the median voter's support) deliver \emph{fewer} earmarks to their districts.
  \end{itemize}
  \vspace{0.2em}
  \textbf{But ``core constituencies'' matter too:}
  \begin{itemize}
    \setlength{\itemsep}{1pt}
    \item \textbf{Levitt (1996):} two senators from the \emph{same} state (same statewide median voter) vote very differently if from different parties --- more like same-party senators from \emph{other} states.
    \item \textbf{Lee et al.\ (2004):} narrowly-elected politicians vote just as ideologically as landslide winners --- not more moderately, as the median voter model would predict.
    \item \textbf{Washington (2008):} legislators with more daughters (random within family size) vote more pro-women's-issues --- personal experience, not just voter demand, shapes behavior.
  \end{itemize}
\end{frame}

\begin{frame}{Application: The Earmark Transparency Backfire (2007)}
  In 2007, House Democratic leaders required earmarks to be transparently linked to their sponsor --- hoping to \emph{shame} politicians into requesting fewer.
  \begin{alertblock}{The opposite happened}
    \emph{``Far from causing embarrassment, the new transparency has raised the value of earmarks as a measure of members' clout''} (\emph{NYT}). Lawmakers competed to attach their names to earmarks and publicized them as a way of claiming credit with constituents.
  \end{alertblock}
  \small Congress eventually banned earmarks outright in 2011 --- but the underlying spending didn't stop, it just became less direct (e.g., routing extra military aircraft orders through district-based manufacturers).
\end{frame}

\begin{frame}{Increasing Polarization in American Politics}
  A defining recent trend: the two US parties have moved steadily further apart.
  \begin{itemize}
    \item House Republicans' measured ideological polarization score has \textbf{more than doubled} since the 1960s (McCarthy et al., updated 2020).
    \item \textbf{``Affective polarization''} --- how much people rate their own party above the other on a feelings thermometer --- rose from a 27-point gap in 1978 to 46 points by 2016.
    \item Among developed democracies, the US shows by far the \textbf{fastest-growing} polarization of the last four decades; several peer countries (UK, Norway, Sweden, Germany, Australia) actually \emph{depolarized} over the same period.
  \end{itemize}
\end{frame}

\begin{frame}{Why Is America Polarizing? Four Explanations}
  \small
  \begin{enumerate}
    \setlength{\itemsep}{3pt}
    \item \textbf{Voter polarization (``tribalism''):} the correlation between self-identified ideology and vote choice has grown sharply since 1972 (Haidt \& Abrams, 2015). Share of Americans opposed to their child marrying someone of the \emph{opposite party}: up from $\sim$5\% in 1960 to 49\% (Republicans) / 33\% (Democrats) today.
    \item \textbf{Media proliferation:} cable news and the internet let voters self-select into partisan ``echo chambers.'' DellaVigna \& Kaplan (2007) and Martin \& Yorukoglu (2017) both find that Fox News's (quasi-random) channel placement shifted local Republican vote shares.
    \item \textbf{Gerrymandering:} discussed next.
    \item \textbf{Other institutional changes:} nationalized campaign fundraising, committee chairs now assigned by party loyalty rather than seniority, and geographic sorting (liberal, educated workers concentrating in cities) all reinforce polarization.
  \end{enumerate}
\end{frame}

\begin{frame}{Gerrymandering: Cracking and Packing}
  \begin{block}{Gerrymandering}
    Manipulating district maps, by the party in power, to maintain or grow that party's power.
  \end{block}
  \begin{itemize}
    \item \textbf{Cracking:} split the opposing party's voters across many districts so they narrowly lose each one.
    \item \textbf{Packing:} concentrate the opposing party's voters into a few districts so they win those by a landslide (and lose everywhere else).
  \end{itemize}
  \small \textbf{Pennsylvania, 2016:} Democrats won 48\% of the statewide popular vote but only 23\% of House seats --- a telltale gerrymandering signature. After a court-ordered, party-neutral redraw, the 2018 result (53\% of the vote) yielded nearly half the seats.
  \vspace{0.3em}
  \begin{block}{Efficiency gap}
    Party A's wasted votes minus Party B's wasted votes, divided by total votes cast --- a standard metric for measuring gerrymandering.
  \end{block}
\end{frame}

\begin{frame}{Malapportionment: The Senate and the Electoral College}
  Even \emph{without} active gerrymandering, US institutions build in a structural skew:
  \begin{itemize}
    \item Every state gets 2 senators, regardless of population. By 2040, an estimated 30\% of Americans (in the 35 least populous states) will elect \textbf{70} of 100 senators --- and these ``over-represented'' states skew more White, male, and rural than the country as a whole.
    \item The Electoral College guarantees every state at least 3 electoral votes: a Wyoming vote (pop.\ 579,000) counts nearly \textbf{4$\times$} a California vote (pop.\ $\approx$40 million) in a presidential election.
  \end{itemize}
\end{frame}

\section{9.4 Public Choice Theory}

\begin{frame}{From Benevolent Government to Public Choice Theory}
  Most of public finance --- and most of this course --- assumes a \textbf{benevolent} government maximizing social welfare. Starting in the 1950s, \textbf{public choice theory} (James Buchanan and Gordon Tullock; Buchanan won the 1986 Nobel) questioned that assumption directly.
  \begin{block}{Government failure}
    The inability or unwillingness of government to act primarily in the interest of its citizens.
  \end{block}
\end{frame}

\begin{frame}{Size-Maximizing Bureaucracy: The Niskanen Model}
  \begin{block}{Niskanen (1971): the budget-maximizing bureaucrat}
    A bureaucrat who runs a monopoly government agency is compensated (salary, staff, status) based on the agency's total output --- so they maximize the \textbf{size} of their budget, not the efficiency of the service they provide.
  \end{block}
  \small Even when the broader government suspects this, it's hard to stop: the bureaucrat knows far more about the agency's \emph{true} cost of providing the service than anyone overseeing them does --- a classic information asymmetry, this time inside government itself.
\end{frame}

\begin{frame}{Leviathan Theory}
  \begin{block}{Brennan \& Buchanan (1980)}
    Treat bureaucrats \emph{and} the broader government as a single monopolist (``Leviathan'') that exploits voters' limited information to maximize the size of the public sector.
  \end{block}
  \begin{itemize}
    \item Explains constitutional and statutory limits that deliberately tie government's hands on taxes and spending (e.g., property-tax caps).
    \item \textbf{Political competition as a check:} Besley et al.\ (2005) study the enfranchisement of Black voters in the 20th-century US South --- areas with more balanced, competitive local elections saw \textbf{25\% higher} long-run growth, partly via lower taxes and better-quality jobs.
  \end{itemize}
\end{frame}

\begin{frame}{Application: Government Corruption Around the World}
  \begin{block}{Corruption}
    Abuse of power by officials to maximize their own wealth or that of their associates.
  \end{block}
  \begin{itemize}
    \item \textbf{Illinois:} Governor George Ryan sold state contracts for cash and gifts (2003); his successor, Rod Blagojevich --- elected as a reformer --- was arrested in 2008 for trying to sell Barack Obama's vacated Senate seat.
    \item \textbf{Malaysia:} PM Najib Razak's 2018 arrest followed a \$4.5B state investment-fund theft, partly laundered through Hollywood film financing and a \$27.3M diamond necklace.
    \item \textbf{Zaire (DRC):} Joseph Mobutu treated the resource-rich country as personal property (1965--1997); per-capita income \textbf{fell by two-thirds} while he amassed an estimated \$5 billion fortune.
  \end{itemize}
\end{frame}

\begin{frame}{Why Does Corruption Exist? Electoral Accountability}
  \small
  \begin{itemize}
    \item \textbf{Persson \& Tabellini (2000):} countries where voters elect \emph{individual} candidates (like the US) have \textbf{less} corruption than proportional systems where voters elect a \emph{party slate} --- individual accountability disciplines politicians more directly.
    \item \textbf{Ferraz \& Finan (2008):} randomized federal anti-corruption audits of Brazilian municipalities --- mayors caught abusing funds were significantly \textbf{less likely to be re-elected}. Voters punish corruption when they can see it.
    \item \textbf{Djankov et al.\ (2002):} countries with more bureaucratic \textbf{red tape} to start a business (2 days in Canada vs.\ 152 in Madagascar) have systematically higher corruption.
  \end{itemize}
\end{frame}

\begin{frame}{Empirical Evidence: Government Failure and Economic Growth}
  \small
  \begin{itemize}
    \item \textbf{Mauro (1995):} nations with more corruption and red tape grow more slowly; if the least efficient country in his sample matched the most efficient, its growth rate would rise by \textbf{4.9 points/year}.
    \item \textbf{Acemoglu, Johnson \& Robinson (2001):} colonies where European settlers avoided disease (and so settled permanently, building strong institutions) grew far faster post-independence than colonies governed ``from afar'' purely for extraction --- despite similar starting conditions.
    \item \textbf{Acemoglu (2003):} North and South Korea were a single, culturally similar region with \emph{equal} incomes in 1950 (\$850). By the 2010s: North Korea \$1,990; South Korea \$39,400 --- institutions alone explain a 20-fold divergence.
  \end{itemize}
  \vspace{0.2em}
  \small After Brazil's 2007 reform curtailed vote-buying, political competition rose --- and health spending grew 10\%/year for three years, with the live birth rate improving over 12.5\%.
\end{frame}

% =============================================================
\begin{frame}{Conclusion}
  \begin{itemize}
    \item Political economy is where the ``why'' question of public finance lives: government is not a single benevolent actor, but a system that must aggregate the preferences of millions of people.
    \item The median voter model --- for both direct and representative democracy --- is a powerful, well-evidenced benchmark, but it rests on assumptions (single-dimensionality, full information, no money, no ideology) that regularly fail in practice.
    \item Public choice theory reminds us that market failure is only half the story: \textbf{government failure} --- bureaucratic empire-building, Leviathan growth, outright corruption --- can be just as real, and just as costly to economic growth.
  \end{itemize}
\end{frame}
